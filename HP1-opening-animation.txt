(function(){
  const splash=document.getElementById("hp1Intro");
  let seen=false;
  try{
    seen=sessionStorage.getItem("hp1IntroSeen")==="1";
    if(!seen)sessionStorage.setItem("hp1IntroSeen","1");
  }catch(e){}
  const close=()=>{
    if(!splash)return;
    document.body.classList.remove("intro-active");
    splash.classList.add("hide");
    setTimeout(()=>splash.remove(),850);
  };
  if(seen){close();return}
  setTimeout(close,2300);
})();
